const functions = require("firebase-functions"); const admin = require("firebase-admin"); admin.initializeApp(); const db = admin.firestore(); // ========== 1. OFFLINE AI LIVE TRADING - EVERY 2 MINUTES - $2 STARTING CAPITAL ========== exports.autoTrading = functions.pubsub.schedule("every 2 minutes").onRun(async (context) => { console.log("🤖 Auto Trading - SolarAI Capital - Delaware, USA - $2 Starting Capital - Trading offline"); let robotRef = db.collection("robotBalances").doc("main"); let robotDoc = await robotRef.get(); if (!robotDoc.exists) { // Initialize with your $2 starting capital await robotRef.set({ robotBalance: 2.00, realProfit: 0, vault: 0, totalProfit: 0, totalTrades: 0, totalDeposits: 2.00, aiActive: true, lastTrade: admin.firestore.FieldValue.serverTimestamp(), createdAt: admin.firestore.FieldValue.serverTimestamp(), startingCapital: 2.00 }); console.log("✅ Initialized Robot with $2 starting capital"); return null; } let data = robotDoc.data(); let rb = parseFloat(data.robotBalance || 0); let vb = parseFloat(data.vault || 0); let pr = parseFloat(data.realProfit || 0); let trades = parseInt(data.totalTrades || 0); if (rb < 0.50) { console.log("⚠️ Robot balance too low $"+rb.toFixed(2)+" - Need at least $0.50 to trade"); return null; } if (data.aiActive === false) { console.log("⏸️ AI Trading paused by admin"); return null; } // REALISTIC AI TRADING LOGIC - 2% to 8% profit per trade (like real trading bot) let profitPercent = 0.02 + Math.random() * 0.06; // 2% to 8% let profit = rb * profitPercent; let toVault = profit * 0.5; let toRobot = profit * 0.5; let newRB = rb + toRobot; let newVB = vb + toVault; let newPR = pr + profit; await robotRef.update({ robotBalance: newRB, vault: newVB, realProfit: newPR, totalProfit: admin.firestore.FieldValue.increment(profit), totalTrades: admin.firestore.FieldValue.increment(1), lastTrade: admin.firestore.FieldValue.serverTimestamp(), lastTradeProfit: profit, lastTradePercent: profitPercent * 100, lastUpdate: admin.firestore.FieldValue.serverTimestamp() }); // Log trade to profitHistory collection await db.collection("profitHistory").add({ profit: profit, percent: profitPercent * 100, robotBalance: newRB, vault: newVB, toVault: toVault, toRobot: toRobot, time: new Date().toLocaleString("en-US", {timeZone: "America/New_York"}), createdAt: admin.firestore.FieldValue.serverTimestamp(), timestamp: Date.now(), auto: true, offline: true }); console.log(`✅ Trade #${trades+1}: +$${profit.toFixed(4)} (${(profitPercent*100).toFixed(2)}%) | Robot $${newRB.toFixed(2)} | Vault $${newVB.toFixed(2)} | Starting $2`); return null; }); // ========== 2. AUTO ROI 1.2% EVERY 24H - EST - SILENT TO INVESTORS ========== exports.dailyROI = functions.pubsub.schedule("0 14 * * *").timeZone("America/New_York").onRun(async (context) => { console.log("🚀 Daily Profit 1.2% - Delaware, USA - EST 2PM"); const transactionsSnap = await db.collection("transactions").where("type", "==", "deposit").where("status", "==", "approved").get(); let totalDistributed = 0; let count = 0; for (let doc of transactionsSnap.docs) { let tx = doc.data(); let amount = parseFloat(tx.amount); if (isNaN(amount) || amount <= 0) continue; let roiAmount = amount * 0.012; let email = tx.email; if (!email) continue; let userRef = db.collection("users").doc(email); let userDoc = await userRef.get(); if (!userDoc.exists) { await userRef.set({ email: email, balance: roiAmount, totalProfit: roiAmount, lastProfitDate: new Date().toISOString(), createdAt: admin.firestore.FieldValue.serverTimestamp() }); } else { let currentBal = parseFloat(userDoc.data().balance || 0); let totalProfit = parseFloat(userDoc.data().totalProfit || 0); await userRef.update({ balance: currentBal + roiAmount, totalProfit: totalProfit + roiAmount, lastProfitDate: new Date().toISOString(), nextProfitTime: Date.now() + 86400000, nextROI: Date.now() + 86400000 }); } await db.collection("roiHistory").add({ email: email, pkg: tx.pkg, deposit: amount, roiAmount: roiAmount, profitAmount: roiAmount, percent: "1.2%", date: new Date().toLocaleString("en-US", {timeZone: "America/New_York"}), createdAt: admin.firestore.FieldValue.serverTimestamp(), auto: true }); let robotRef = db.collection("robotBalances").doc("main"); let robotDoc = await robotRef.get(); if (robotDoc.exists) { let rb = parseFloat(robotDoc.data().robotBalance || 0); let pr = parseFloat(robotDoc.data().realProfit || 0); let vb = parseFloat(robotDoc.data().vault || 0); let tr = parseFloat(robotDoc.data().totalProfit || 0); await robotRef.update({ realProfit: pr + roiAmount, vault: vb + (roiAmount * 0.5), robotBalance: rb + (roiAmount * 0.5), totalProfit: tr + roiAmount, lastUpdate: admin.firestore.FieldValue.serverTimestamp() }); } totalDistributed += roiAmount; count++; } console.log(`✅ Profit Distributed: ${count} investors, Total $${totalDistributed.toFixed(2)} at 1.2% - Delaware EST 2PM`); await db.collection("auditLogs").add({ action: `AUTO PROFIT 1.2% - ${count} investors - $${totalDistributed.toFixed(2)} - ${new Date().toLocaleString("en-US", {timeZone: "America/New_York"})} EST - Delaware, USA`, count: count, total: totalDistributed, date: new Date().toLocaleString("en-US", {timeZone: "America/New_York"}), createdAt: admin.firestore.FieldValue.serverTimestamp() }); return null; }); // ========== 3. ON DEPOSIT APPROVED - LINK TO ROBOT ========== exports.onDepositApproved = functions.firestore.document("transactions/{txId}").onUpdate(async (change, context) => { const before = change.before.data(); const after = change.after.data(); if (before.status!== "approved" && after.status === "approved" && after.type === "deposit") { let amount = parseFloat(after.amount); let email = after.email; let nextProfit = Date.now() + 86400000; let userRef = db.collection("users").doc(email); let userDoc = await userRef.get(); if (!userDoc.exists) { await userRef.set({ email: email, balance: amount, initialDeposit: amount, nextProfitTime: nextProfit, nextROI: nextProfit, pkg: after.pkg, createdAt: admin.firestore.FieldValue.serverTimestamp() }); } else { let currentBal = parseFloat(userDoc.data().balance || 0); await userRef.update({ balance: currentBal + amount, nextProfitTime: nextProfit, nextROI: nextProfit, lastDeposit: admin.firestore.FieldValue.serverTimestamp() }); } let robotRef = db.collection("robotBalances").doc("main"); let robotDoc = await robotRef.get(); if (!robotDoc.exists) { await robotRef.set({ robotBalance: amount, totalDeposits: amount, createdAt: admin.firestore.FieldValue.serverTimestamp() }); } else { if (!before.linkedToAI) { let rb = parseFloat(robotDoc.data().robotBalance || 0); let td = parseFloat(robotDoc.data().totalDeposits || 0); await robotRef.update({ robotBalance: rb + amount, totalDeposits: td + amount, lastDeposit: amount, lastDepositEmail: email }); } } } return null; }); // ========== 4. MANUAL TRIGGER ROI ========== exports.triggerROIManual = functions.https.onCall(async (data, context) => { const transactionsSnap = await db.collection("transactions").where("type", "==", "deposit").where("status", "==", "approved").get(); let total = 0; for (let doc of transactionsSnap.docs) { let tx = doc.data(); let roiAmount = parseFloat(tx.amount) * 0.012; let userRef = db.collection("users").doc(tx.email); let userDoc = await userRef.get(); let currentBal = userDoc.exists? parseFloat(userDoc.data().balance || 0) : 0; if (userDoc.exists) { await userRef.update({ balance: currentBal + roiAmount, totalProfit: admin.firestore.FieldValue.increment(roiAmount), nextProfitTime: Date.now() + 86400000, nextROI: Date.now() + 86400000 }); } await db.collection("roiHistory").add({ email: tx.email, pkg: tx.pkg, deposit: tx.amount, roiAmount: roiAmount, profitAmount: roiAmount, date: new Date().toLocaleString("en-US", {timeZone: "America/New_York"}), createdAt: admin.firestore.FieldValue.serverTimestamp(), manual: true }); total += roiAmount; } return { success: true, total: total, count: transactionsSnap.size, message: `Daily Profit $${total.toFixed(2)} credited to ${transactionsSnap.size} investors - Delaware, USA` }; }); // ========== 5. ADMIN WITHDRAW REAL PROFIT - AFTER 2 HOURS ========== exports.adminWithdrawProfit = functions.https.onCall(async (data, context) => { let amount = parseFloat(data.amount); let wallet = data.wallet; let network = data.network || "USDT TRC20"; if (isNaN(amount) || amount <= 0) { throw new functions.https.HttpsError("invalid-argument", "Invalid amount"); } let robotRef = db.collection("robotBalances").doc("main"); let robotDoc = await robotRef.get(); if (!robotDoc.exists) { throw new functions.https.HttpsError("not-found", "Robot balance not found - Start with $2 first"); } let pr = parseFloat(robotDoc.data().realProfit || 0); let vb = parseFloat(robotDoc.data().vault || 0); let rb = parseFloat(robotDoc.data().robotBalance || 0); if (amount > pr) { throw new functions.https.HttpsError("failed-precondition", `Insufficient profit. Available profit: $${pr.toFixed(2)}, Vault: $${vb.toFixed(2)}, Robot: $${rb.toFixed(2)}`); } // Deduct from profit and vault await robotRef.update({ realProfit: pr - amount, vault: vb - (amount * 0.5), robotBalance: rb - (amount * 0.5), lastWithdraw: amount, lastWithdrawWallet: wallet, lastWithdrawTime: admin.firestore.FieldValue.serverTimestamp() }); // Log withdrawal await db.collection("adminWithdrawals").add({ amount: amount, wallet: wallet, network: network, date: new Date().toLocaleString("en-US", {timeZone: "America/New_York"}), createdAt: admin.firestore.FieldValue.serverTimestamp(), robotBalanceBefore: rb, profitBefore: pr, vaultBefore: vb }); return { success: true, message: `✅ Withdrawal $${amount.toFixed(2)} to ${network} ${wallet} processed - Real profit from $2 starting capital trading offline`, remainingProfit: pr - amount, robotBalance: rb - (amount * 0.5) }; }); // ========== 6. TOGGLE AI TRADING ON/OFFLINE ========== exports.toggleAITrading = functions.https.onCall(async (data, context) => { let active = data.active; // true or false let robotRef = db.collection("robotBalances").doc("main"); await robotRef.update({ aiActive: active, lastToggle: admin.firestore.FieldValue.serverTimestamp() }); return { success: true, aiActive: active, message: active ? "✅ AI Trading Active - Trading offline every 2 minutes - $2 capital growing" : "⏸️ AI Trading Paused - Offline trading stopped" }; });
